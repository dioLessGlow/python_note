15. 计算 10! 结果末尾有多少个连续的 0

查看答案

for i in range(1, 11):
    result *= i
count = 0
for c in reversed(str(result)):
    if c == "0":
        count += 1
    else:
        break
print(count)  # 2

4. 统计 "abracadabra" 中 a 和 b 的次数

查看答案

# 或循环统计
sum_a = 0; sum_b = 0
for i in a:
    if i == 'a': sum_a += 1
    elif i == 'b': sum_b += 1
print(sum_a, sum_b)  # 5 2

else:
    print("n")
