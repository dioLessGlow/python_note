1. 使用 fromkeys() 创建一个字典，键为 ['a','b','c']，默认值为空列表 []。观察三个键对应的值是否为同一个列表对象（可用 id() 或 is 验证），并解释原因。写出正确的初始化方式，使每个键拥有独立的空列表。

查看答案
# fromkeys：所有键共享同一个可变对象
a = ['a','b','c']
b = dict.fromkeys(a, [])
print(b)                     # {'a': [], 'b': [], 'c': []}
print(b['a'] is b['b'])      # True ← 是同一个列表！
b['a'].append(1)
print(b)                     # {'a': [1], 'b': [1], 'c': [1]}

# ✅ 正确做法：字典推导式，每个键独立创建 []
b = {k: [] for k in a}
print(b['a'] is b['b'])      # False ← 各自独立

2. 给定字典 prices = {'apple': 3.5, 'banana': None, 'cherry': 4.0}，使用 get() 获取 'banana' 的价格，若值为 None 则返回 0.0；同时使用 setdefault() 为不存在的键 'date' 设置默认值 5.0 并获取。

查看答案
prices = {'apple': 3.5, 'banana': None, 'cherry': 4.0}

# get()：取值，不存在返回默认值（不修改原字典）
price = prices.get('banana')
if price is None:
    price = 0.0
print(price)                 # 0.0

# setdefault()：键不存在时插入默认值并返回
prices.setdefault('date', 5)
print(prices)                # {'apple': 3.5, 'banana': None, 'cherry': 4.0, 'date': 5}

3. 合并两个字典 dict_a = {'x': 1, 'y': 2} 和 dict_b = {'y': 3, 'z': 4}，要求保留 dict_b 的值覆盖 dict_a。分别使用 update() 方法、| 运算符（Python 3.9+）和 {**dict_a, **dict_b} 解包三种方式实现。

查看答案
dict_a = {'x': 1, 'y': 2}
dict_b = {'y': 3, 'z': 4}

# 1. update（修改原字典）
dict_c = dict_a.copy()
dict_c.update(dict_b)
print(dict_c)                # {'x': 1, 'y': 3, 'z': 4}

# 2. | 运算符（Python 3.9+）
dict_c = dict_a | dict_b
print(dict_c)                # {'x': 1, 'y': 3, 'z': 4}

# 3. ** 解包
dict_c = {**dict_a, **dict_b}
print(dict_c)                # {'x': 1, 'y': 3, 'z': 4}

8. 使用字典推导式生成一个字典，键为 1 到 10 的整数，值为该数的阶乘（如 {1:1, 2:2, 3:6, ...}），要求不使用 math.factorial，用推导式内部嵌套循环实现。

查看答案
from functools import reduce
import operator

# 方法1：推导式 + reduce
a = {k: reduce(operator.mul, range(1, k + 1), 1) for k in range(1, 11)}
print(a)
# {1:1, 2:2, 3:6, 4:24, 5:120, 6:720, 7:5040, 8:40320, 9:362880, 10:3628800}

# 方法2：嵌套循环
result = {}
for k in range(1, 11):
    fact = 1
    for i in range(1, k + 1):
        fact *= i
    result[k] = fact
print(result)

15. 统计字典 words = {'hello': 5, 'world': 5, 'python': 6, 'code': 4} 中值为偶数的键值对数量，并原地删除所有值小于 5 的键值对（注意遍历时直接删除的陷阱，写出正确做法）。

查看答案
words = {'hello': 5, 'world': 5, 'python': 6, 'code': 4}

# 统计值为偶数的数量
total = sum(1 for v in words.values() if v % 2 == 0)
print(total)                 # 2

# ❌ 遍历时直接删除 → RuntimeError: dictionary changed size during iteration
# ✅ 先收集要删的键，再删除
to_del = [k for k, v in words.items() if v < 5]
for k in to_del:
    del words[k]
print(words)                 # {'hello': 5, 'world': 5, 'python': 6}

17. 实现一个递归函数 deep_upper(obj)，将多层嵌套字典中的所有字符串值转换为大写（如 {'a': {'b': 'hello'}} → {'a': {'b': 'HELLO'}}）；若值是列表，列表内的字典也需递归处理。

查看答案
def deep_upper(obj):
    for k, v in obj.items():
        if type(v) is str:
            obj[k] = v.upper()
        elif type(v) is list:
            for key, value in enumerate(v):
                if type(value) is str:
                    v[key] = value.upper()
                elif type(value) is dict:
                    deep_upper(value)
        elif type(v) is dict:
            deep_upper(v)
    return obj

data = {
    'name': 'alice',
    'tags': ['python', {'label': 'coding'}, 'java'],
    'info': {'city': 'beijing', 'hobbies': [{'sport': 'basketball'}, 'reading']}
}
print(deep_upper(data))
# 所有字符串递归转大写

19. 使用 collections.defaultdict(list) 将列表 [('a', 1), ('b', 2), ('a', 3), ('c', 4), ('b', 5)] 分组为 {'a': [1, 3], 'b': [2, 5], 'c': [4]}，并说明 defaultdict 与普通 dict 在访问缺失键时的行为差异。

查看答案
from collections import defaultdict

a = [('a', 1), ('b', 2), ('a', 3), ('c', 4), ('b', 5)]

# defaultdict：缺失键时自动调用 list() 生成空列表
grouped = defaultdict(list)
for key, value in a:
    grouped[key].append(value)
print(dict(grouped))         # {'a': [1, 3], 'b': [2, 5], 'c': [4]}

# 普通 dict 访问缺失键 → KeyError
# defaultdict 访问缺失键 → 自动插入默认值并返回

# 手动实现（不用 defaultdict）
new_dict = {}
for k, v in a:
    if k not in new_dict:
        new_dict[k] = []
    new_dict[k].append(v)
print(new_dict)              # 同上

20. 实现一个不使用 copy 模块的 deep_copy_dict 函数，能够递归处理嵌套字典、列表和基本数据类型（int, str, bool 等）的深拷贝。用包含嵌套字典和列表的复杂结构验证：修改副本不影响原对象。

查看答案
def deep_copy_dict(obj):
    if type(obj) is dict:
        return {k: deep_copy_dict(v) for k, v in obj.items()}
    elif type(obj) is list:
        return [deep_copy_dict(item) for item in obj]
    elif type(obj) is tuple:
        return tuple(deep_copy_dict(item) for item in obj)
    elif type(obj) is set:
        return {deep_copy_dict(item) for item in obj}
    else:
        return obj

# 验证
original = {'a': [1, 2, {'b': 3}], 'c': {'d': [4, 5]}}
copied = deep_copy_dict(original)

copied['a'].append(99)
copied['a'][2]['b'] = 999

print(original)  # {'a': [1, 2, {'b': 3}], 'c': {'d': [4, 5]}}
print(copied)    # {'a': [1, 2, {'b': 999}, 99], 'c': {'d': [4, 5]}}
# 修改副本不影响原对象 ✅
