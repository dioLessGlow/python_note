在 BankAccount 类中添加实例方法 deposit(amount) 和 withdraw(amount)。deposit 增加余额；withdraw 减少余额，若余额不足则抛出 ValueError("余额不足")。要求：创建实例，依次存入 500、取出 200、再尝试取出 400，观察异常。

class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance
    def deposit(self,amount):
        if amount <= 0:
            raise ValueError("存款金额必须大于0")
        self.balance += amount
        print(f"存款成功，当前余额剩余：{self.balance}")
    def withdraw(self, amount):
        if self.balance -amount >= 0:
            self.balance -= amount
            print(f"取款成功，当前余额剩余：{self.balance}")
        else:
            raise ValueError("余额不足")

class BankAccount:

bank_name = "Python Bank"

interest_rate = 0.015

    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance
    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("存款金额必须大于0")

    def withdraw(self, amount):

            raise ValueError("余额不足")

@classmethod

    def from_dict(cls, data):
        return cls(data['owner'], data.get('balance', 0))

@staticmethod

    def validate_account_number(num):
        return isinstance(num,str) and len(num) == 10 and num.isdigit()
account = BankAccount.from_dict({'owner': 'Alice', 'balance': 1000})
print(account.owner)
print(account.balance)
