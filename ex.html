二、join vs + —— 何时用哪个

+：少量拼接，直观

# ✅ 少量字符串
a = "Na"
b = a * 4 + " Batman!"   # "NaNaNaNa Batman!"

# ✅ 少数变量拼接
full = first + " " + last

# ❌ 循环中用 + 拼接大量字符串（效率低！）
result = ""
for s in big_list:          # 每次循环创建新字符串
    result += s             # O(n²) 时间复杂度

join：列表→字符串，大量拼接

# ✅ 列表转字符串（推荐）
words = ["Python", "is", "powerful"]
print(' '.join(words))        # "Python is powerful"

# ✅ 循环中收集到列表，最后 join（高效！）
parts = []
for s in big_list:
    parts.append(s)
result = ''.join(parts)        # O(n) 一次拼接

# ✅ join + 推导式（最简洁）
result = ''.join(s for s in big_list if s)

对比总结：

操作                    用 join                    用 +
──────────────────────────────────────────────────
列表→字符串              ✅ 'sep'.join(list)        ❌ 不行
少量字符串拼接            ❌ 没必要                 ✅ a + b
循环中大量拼接            ✅ join(列表)             ❌ += 效率低
数字→字符串拼接           ❌ 需先转str              ✅ str(n) + "元"
混合类型                  ❌ 需全转str              ✅ str(x) + str(y)

六、编程题代码参考

# 1. 素数判断（O(√n)）
def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

# 2. 递归反转字符串
def reverse_string(s):
    if len(s) <= 1:
        return s
    return s[-1] + reverse_string(s[:-1])

# 3. 统计元素出现次数
def count_elements(lst):
    return {x: lst.count(x) for x in set(lst)}

# 4. 快速排序（O(n log n) 平均 / O(n²) 最坏）
def quick_sort(arr):
    if len(arr) <= 1:
        return arr
    pivot = arr[len(arr) // 2]
    left   = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right  = [x for x in arr if x > pivot]
    return quick_sort(left) + middle + quick_sort(right)

# 5. 装饰器统计调用次数
from functools import wraps

def call_counter(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        wrapper.call_count += 1
        return func(*args, **kwargs)
    wrapper.call_count = 0
    return wrapper

@call_counter
def greet(name):
    return f"Hello, {name}!"

print(greet("Alice"))      # Hello, Alice!
print(greet.call_count)    # 1

def deep_upper(obj):
    for k, v in obj.items():
        if type(v) is str:
            obj[k] = v.upper()
        elif type(v) is list:
            for key, value in enumerate(v):
                if type(value) is str:
                    v[key] = value.upper()
                elif type(value) is dict:
                    deep_upper(value)
        elif type(v) is dict:
            deep_upper(v)
    return obj

data = {

    'tags': ['python', {'label': 'coding'}, 'java'],

十九、递归深拷贝（支持 dict/list/tuple/set）

def deep_copy_dict(obj):
    if type(obj) is dict:
        return {k: deep_copy_dict(v) for k, v in obj.items()}
    elif type(obj) is list:
        return [deep_copy_dict(item) for item in obj]
    elif type(obj) is tuple:
        return tuple(deep_copy_dict(item) for item in obj)
    elif type(obj) is set:
        return {deep_copy_dict(item) for item in obj}
    else:
        return obj

一、类属性 vs 实例属性

# 类属性：所有实例共享，修改影响所有实例
class BankAccount:
    bank_name = "Python Bank"    # 类属性
    interest_rate = 0.015        # 类属性

# 实例属性：每个实例独立
    def __init__(self, owner, balance=0):
        self.owner = owner        # 实例属性
        self._balance = balance   # 实例属性

# 修改类属性 → 所有实例受影响
BankAccount.interest_rate = 0.02  # ✅
# 修改实例属性 → 只影响该实例
account1.interest_rate = 0.03    # ❌ 创建了新实例属性，遮蔽类属性

二、self vs __init__

# self：当前实例的引用
# __init__：构造方法，创建实例时自动调用

class Dog:
    def __init__(self, name):
        self.name = name      # self.属性 = 参数

    def bark(self):
        print(f"{self.name} 叫")  # 使用 self.方法()

dog = Dog("旺财")
dog.bark()

三、@property (getter/setter)

# 私有属性：用单下划线 _ 或双下划线 __ 前缀
self._balance   # 受保护（约定）
self.__pin      # 私有（名称重整 _类名__属性）

# @property：将方法转为属性（只读）
class BankAccount:
    def __init__(self):
        self._balance = 0

    @property
    def balance(self):
        return self._balance

# @setter：控制属性赋值
    @balance.setter
    def balance(self, value):
        if value < 0:
            self._balance = 0
        else:
            self._balance = value

account.balance = -100  # 自动调用 setter
print(account.balance)   # 自动调用 getter → 0

四、继承与 super()

# 继承：子类获得父类所有属性和方法
class SavingsAccount(BankAccount):
    pass

# super()：调用父类方法
class SavingsAccount(BankAccount):
    def __init__(self, owner, balance, interest_rate):
        super().__init__(owner, balance)  # 调用父类 __init__
        self.interest_rate = interest_rate

    def deposit(self, amount):
        super().deposit(amount)  # 调用父类方法
        # 子类自定义逻辑

五、@staticmethod（静态方法）

# 不需要 self 或 cls，相当于普通函数
# 通常用于工具函数、与类逻辑相关的辅助功能

class BankAccount:
    @staticmethod
    def validate_account_number(num):
        return isinstance(num, str) and len(num) == 10 and num.isdigit()

# 调用：不需要实例
BankAccount.validate_account_number("1234567890")

六、@classmethod（类方法）

# 第一个参数是类本身(cls)，用于创建实例、替代构造函数

class BankAccount:
    @classmethod
    def from_dict(cls, data):
        return cls(data['owner'], data.get('balance', 0))

# 调用：不需要实例，直接用类名
account = BankAccount.from_dict({'owner': 'Alice', 'balance': 1000})

七、私有属性与名称重整

# 单下划线 _：受保护，约定不直接访问
self._balance

# 双下划线 __：私有，触发名称重整(name mangling)
self.__pin  # 实际存储为 _BankAccount__pin

# 提供 getter/setter 访问
@property
def pin(self):
    return self.__pin

@pin.setter
def pin(self, value):
    if len(value) != 6:
        raise ValueError("PIN 必须是6位")
    self.__pin = value

八、魔术方法汇总

# 字符串表示
__str__(self)   # print(obj) → 用户友好
__repr__(self)  # repr(obj) → 开发者友好，eval(repr(obj)) 可重建

# 比较运算
__eq__(self, other)    # ==
__lt__(self, other)    # <
__le__(self, other)    # <=
__gt__(self, other)    # >
__ge__(self, other)    # >=

# 算术运算
__add__(self, other)   # +
__sub__(self, other)   # -
__mul__(self, other)   # *

# 容器相关
__len__(self)          # len(obj)
__getitem__(self, key) # obj[key]
__setitem__(self, key, value)  # obj[key] = value
__call__(self, ...)    # obj() 可调用

# 上下文管理
__enter__(self)       # with obj as x:
__exit__(self, ...)   # 退出with块

九、抽象基类 (abc)

from abc import ABC, abstractmethod

class PaymentMethod(ABC):
    @abstractmethod
    def pay(self, amount):
        pass

    @abstractmethod
    def refund(self, amount):
        pass

# 子类必须实现所有抽象方法，否则无法实例化
class CreditCard(PaymentMethod):
    def pay(self, amount):
        print(f"支付 {amount}")

十、类属性计数

class Transaction:
    _count = 0  # 类属性，所有实例共享

    def __init__(self, ...):
        Transaction._count += 1  # 每次创建实例计数+1

    @classmethod
    def get_count(cls):
        return cls._count
