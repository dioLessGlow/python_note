一、requests 请求网页/接口

import requests

# GET 请求（相当于浏览器按回车）
response = requests.get("https://www.baidu.com/")

# 查看状态码：200成功 / 404找不到 / 500服务器错误
print("状态码:", response.status_code)

# 查看返回内容
print("网页内容:", response.text)

# 中文乱码时，手动指定编码
response.encoding = "utf-8"

# 如果返回的是 JSON 数据，直接用 .json() 转成字典/列表
data = response.json()

二、请求头 headers（Token/Cookie/UA）

import requests

# F12 → Network → Headers 里复制，防止被识别为爬虫
headers = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
    "Accept": "application/json, text/plain, */*",
    # 登录凭证二选一：
    "Authorization": "Bearer 你的Token",   # 方式A：Bearer Token（推荐）
    "Cookie": "session=xxx; token=yyy",     # 方式B：Cookie
}

# 需要登录的接口用 GET 或 POST
response = requests.get(API_URL, headers=headers)
response = requests.post(API_URL, headers=headers, json=PAYLOAD)

四、表格数据提取

from bs4 import BeautifulSoup

# HTML 表格结构：tr=行, th=表头, td=单元格
# <table>
#   <tr><th>场景</th><th>FPS</th><th>温度</th></tr>
#   <tr><td>娃娃机</td><td>60</td><td>65</td></tr>
#   ...
soup = BeautifulSoup(html_doc, "lxml")
rows = soup.find_all("tr")               # 所有行

data = []
for row in rows:
    cells = row.find_all("td")           # 行的所有单元格
    if cells:                            # 跳过表头那行（没有 td 只有 th）
        scene = cells[0].text            # 第1列
        fps   = cells[1].text            # 第2列
        temp  = cells[2].text            # 第3列
        data.append([scene, fps, temp])

print(data)  # [['娃娃机','60','65'], ['形象投屏','45','72']]

八、数据清洗与字段适配

import ast
import pandas as pd

# 1. 字段名大小写不固定 → 自动匹配
def find_column(candidates, df_cols):
    for c in candidates:
        matches = [col for col in df_cols if c.lower() == col.lower()]
        if matches:
            return matches[0]
    return None

fps_col = find_column(["FPS", "fps", "Fps"], df.columns)

# 2. 字段可能是字符串列表 "[16,17,18]" → 安全转成真列表
def to_list(value):
    if isinstance(value, str):
        try:
            return ast.literal_eval(value)      # 安全解析 "[16,17,18]"
        except Exception:
            return [float(x) for x in value.strip("[]").split(",") if x.strip()]
    if isinstance(value, list):
        return value
    return [float(value)]

ftime_list = to_list(row["FTime"])
high_frames = [t for t in ftime_list if t > 18]   # 帧耗时 > 18ms 的帧
